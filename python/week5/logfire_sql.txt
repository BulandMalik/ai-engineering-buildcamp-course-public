SELECT
  trace_id,
  min(start_timestamp) AS started,
  count(*) AS spans,
  max(CASE
    WHEN span_name = 'trivia_session' THEN 1
    ELSE 0
  END) AS grouped_session
FROM records
WHERE service_name = 'trivia-agent'
GROUP BY trace_id
ORDER BY started DESC
LIMIT 30
